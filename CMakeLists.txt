cmake_minimum_required(VERSION 3.16)
project(neoomen-editor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTKMM REQUIRED IMPORTED_TARGET gtkmm-3.0)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
add_library(prj src/prj.cpp src/m3d.cpp)
target_include_directories(prj PUBLIC src)
target_compile_options(prj PRIVATE -Wall -Wextra -Wpedantic)
add_executable(neoomen-editor src/main.cpp src/editor.cpp src/scene.cpp)
target_link_libraries(neoomen-editor PRIVATE prj PkgConfig::GTKMM PkgConfig::EPOXY)
target_compile_options(neoomen-editor PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(prj_tests tests/prj_tests.cpp)
  target_link_libraries(prj_tests PRIVATE prj)
  add_test(NAME prj_tests COMMAND prj_tests)
  add_executable(m3d_tests tests/m3d_tests.cpp)
  target_link_libraries(m3d_tests PRIVATE prj)
  add_test(NAME m3d_tests COMMAND m3d_tests)
  add_executable(editor_smoke tests/editor_smoke.cpp src/editor.cpp src/scene.cpp)
  target_link_libraries(editor_smoke PRIVATE prj PkgConfig::GTKMM PkgConfig::EPOXY)
  add_executable(scene_smoke tests/scene_smoke.cpp src/editor.cpp src/scene.cpp)
  target_link_libraries(scene_smoke PRIVATE prj PkgConfig::GTKMM PkgConfig::EPOXY)
  # Run manually with a display; it exercises real GTK events and controls.
endif()
install(TARGETS neoomen-editor RUNTIME DESTINATION bin)

option(NEOOMEN_BUILD_GAME "Build the SDL2 Dark Omen game reconstruction" ON)
if(NEOOMEN_BUILD_GAME)
  add_subdirectory(game)
endif()
