cmake_minimum_required(VERSION 3.16)
project(neoomen VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
if(NOT TARGET prj)
  add_library(prj ../src/prj.cpp ../src/m3d.cpp)
  target_include_directories(prj PUBLIC ../src)
endif()
add_library(neoomen_core src/formats.cpp src/battle.cpp src/executable.cpp src/campaign.cpp src/ctl.cpp src/animation.cpp)
target_include_directories(neoomen_core PUBLIC src)
target_link_libraries(neoomen_core PUBLIC prj)
target_compile_options(neoomen_core PRIVATE -Wall -Wextra -Wpedantic)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
add_library(neoomen_movie src/movie.cpp)
target_link_libraries(neoomen_movie PUBLIC neoomen_core PkgConfig::SDL2 PkgConfig::FFMPEG)
target_compile_options(neoomen_movie PRIVATE -Wall -Wextra -Wpedantic)
add_executable(neoomen src/main.cpp src/cursor.cpp src/renderer.cpp src/audio.cpp src/frontend.cpp src/game_ui.cpp)
target_link_libraries(neoomen PRIVATE neoomen_core neoomen_movie PkgConfig::SDL2 PkgConfig::EPOXY)
set_target_properties(neoomen PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_options(neoomen PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(neoomen_tests tests/core_tests.cpp)
  target_link_libraries(neoomen_tests PRIVATE neoomen_core)
  add_test(NAME neoomen_core COMMAND neoomen_tests)
endif()
install(TARGETS neoomen RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(neoomen_runtime_tests tests/runtime_tests.cpp)
  target_link_libraries(neoomen_runtime_tests PRIVATE neoomen_core)
  set(NEOOMEN_GAME_DATA "" CACHE PATH "Original Dark Omen installation for optional integration tests")
  if(NEOOMEN_GAME_DATA)
    add_test(NAME neoomen_original_runtime COMMAND neoomen_runtime_tests "${NEOOMEN_GAME_DATA}" "${CMAKE_CURRENT_BINARY_DIR}/runtime-test.save")
  endif()
endif()
if(BUILD_TESTING)
  add_executable(neoomen_movie_tests tests/movie_tests.cpp)
  target_link_libraries(neoomen_movie_tests PRIVATE neoomen_movie)
  if(NEOOMEN_GAME_DATA)
    add_test(NAME neoomen_movie_intro COMMAND neoomen_movie_tests "${NEOOMEN_GAME_DATA}/Movies/INTRO.TGQ")
    add_test(NAME neoomen_movie_logo COMMAND neoomen_movie_tests "${NEOOMEN_GAME_DATA}/Movies/ENG.TGQ")
  endif()
endif()
if(BUILD_TESTING)
  # Run explicitly on a machine with a display/OpenGL context.
  add_executable(neoomen_frontend_tests tests/frontend_tests.cpp src/cursor.cpp src/frontend.cpp src/renderer.cpp src/game_ui.cpp src/audio.cpp)
  target_link_libraries(neoomen_frontend_tests PRIVATE neoomen_movie PkgConfig::EPOXY)
endif()
if(BUILD_TESTING)
  add_executable(neoomen_audio_tests tests/audio_tests.cpp src/audio.cpp)
  target_link_libraries(neoomen_audio_tests PRIVATE neoomen_core PkgConfig::SDL2)
  add_test(NAME neoomen_audio COMMAND neoomen_audio_tests)
  if(NEOOMEN_GAME_DATA)
    add_test(NAME neoomen_original_audio COMMAND neoomen_audio_tests "${NEOOMEN_GAME_DATA}")
  endif()
endif()
